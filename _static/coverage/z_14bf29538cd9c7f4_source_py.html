35 model_name: str, 

36 tokenizer: Optional[Any] = None, 

37 dtype: Optional[torch.dtype] = None, 

38 gpu_memory_utilization: float = 0.5,

43) -> RemoteBridge: 

44 """Boot a model via vLLM and wrap it in a :class:`RemoteBridge` via :class:`VLLMDriver`. 

45 

46 vLLM drives the forward pass (PagedAttention + ``torch.compile`` + CUDA graphs). 

47 Capture buffers are populated by hooks the plugin installs pre-compile inside 

48 the worker; they come back via ``collective_rpc`` and replay through the 

49 bridge's HookPoint tree. 

50 

51 **Scope vs vllm-lens:** vllm-lens is observation-only. This source extends to 

52 observation + spec-vocabulary *mutation* — each capture hook also applies an 

53 affine transform ``output = output * scale + bias`` (default identity), so 

54 interventions (``suppress`` / ``scale`` / ``add`` / ``set``) propagate to 

55 downstream layers. The hook's return value replaces the module output per 

56 PyTorch ``register_forward_hook`` semantics. **The mutation path under 

57 torch.compile + CUDA graphs is verified end-to-end by** 

58 ``demos/vLLM_Bridge_Integration_Test.ipynb``; unit tests cover the dispatch 

59 protocol only. 

60 

61 Some captures use vLLM-native conventions that differ from HF/HT; see 

62 :mod:`transformer_lens.model_bridge.sources.vllm.overlays.decoder_only` for 

63 which hooks diverge and the conversion to apply for HT-equivalent values. 

64 

65 **Returned logits are log-probs, not raw logits.** vLLM bypasses ``lm_head``; 

66 the driver fills the final position from the sampler's ``log_softmax`` output. 

67 Correct for argmax / next-token, but absolute scale is off — temperature 

68 scaling and logit-lens magnitude analysis will be wrong. Only the final 

69 position is populated (earlier positions are ``-inf``); ``return_type`` in 

70 ``{"loss", "both"}`` is rejected for that reason. 

71 

72 GPU memory cost: each capture buffer is ``max_num_batched_tokens × width`` at 

73 the model's dtype. For Llama-3.2-1B at fp16 with ``max_num_batched_tokens=2048``, 

74 the unembed buffer alone is ~525 MB (2048 × 128256 × 2 bytes); residual-stream 

75 buffers add ~8 MB per hook. The affine intervention hook also allocates a 

76 transient output-shape tensor per forward (even in identity mode), so peak 

77 forward memory is ~1.5× the capture buffers' resident size. 

78 

79 KV-cache footprint: vLLM reserves KV cache sized for ``max_model_len`` × layers 

80 × heads × head_dim. If ``max_model_len`` is left as ``None``, vLLM uses the 

81 model's native context (e.g. 131072 for Llama-3.2-1B) — easily 4+ GiB even 

82 on a 1B model. Pass an explicit ``max_model_len`` (e.g. ``2048`` for typical 

83 mech-interp prompts) to keep the budget on smaller GPUs. 

84 

85 ``enable_batching`` switches to the eager batched path (``enforce_eager``, 

86 ``batch_size > 1``) — the throughput path for SAE/probe data collection. 

87 Default ``False`` keeps the compile-validated single-prompt path. Batched 

88 caches are right-padded with zeros to the longest sequence. 

89 """

95 # which dot-paths to hook before LLM(...) constructs the worker.

126 # Batched capture reads query_start_loc from the forward context, untraceable 

127 # under torch.compile — so the batched path must run eager. 

128 eager_kwargs: Dict[str, Any] = {"enforce_eager": True} if enable_batching else {} 

129 

130 llm = LLM( 

131 model=model_name, 

132 gpu_memory_utilization=gpu_memory_utilization, 

133 max_model_len=max_model_len, 

134 # Critical: vLLM defaults max_num_batched_tokens to 8192 for chunked prefill. 

135 # We size our capture buffer to this same value, so vLLM must compile for 

136 # the matching dynamic-shape range — otherwise Dynamo's symbolic-shape 

137 # hint exceeds the buffer dim and narrow() fails at compile time. 

138 max_num_batched_tokens=max_num_batched_tokens, 

139 # Register TLWorkerExtension so its tl_* methods are reachable via 

140 # collective_rpc. Passed as a dotted path; vLLM imports the class at 

141 # worker construction and mixes it into the Worker via multiple 

142 # inheritance (asserts no attribute name collisions — hence the tl_ prefix). 

143 worker_extension_cls="transformer_lens.model_bridge.sources.vllm.worker_extension.TLWorkerExtension", 

144 # Allow full-vocab logprobs so the driver can synthesize real logits for 

145 # the generated position (vLLM caps logprobs to this value; default 20 is 

146 # too small for mech-interp). One ~512 KB buffer per call; negligible. 

147 max_logprobs=hf_config_preview.vocab_size, 

148 dtype=str(resolved_dtype).replace("torch.", "") if dtype is not None else "auto", 

149 **eager_kwargs, 

150 **_LOCKED_KWARGS, 

151 **vllm_kwargs, 

152 ) 

153 

154 # Capture-path validity is enforced inside patched_load_model during 

155 # LLM(...) above — any missing dot-path raises AttributeError there. By 

156 # the time we reach this line every spec has already been walked successfully. 

157 hf_config = extract_hf_config(llm) 

158 # _config has been consumed into per-Worker state; clear so a non-TL 

159 # vllm.LLM(...) in the same process doesn't inherit our specs. 

160 plugin._config.clear() 

161 if tokenizer is None: 

162 tokenizer = AutoTokenizer.from_pretrained(model_name, token=hf_token) 

163 

164 # Build the adapter (RemoteBridge skips adapter.prepare_model — there's no 

165 # local model tree to walk). Use the shared HF→TL config builder so 

166 # bridge_config is a real TransformerBridgeConfig with all dataclass 

167 # defaults (d_vocab_out=-1, etc.) — not a deep-copied HF config with 

168 # extra fields, which is missing the TL-only attributes BridgeCore reads. 

169 bridge_config = build_bridge_config_from_hf(hf_config, architecture, model_name, resolved_dtype) 

170 adapter = ArchitectureAdapterFactory.select_architecture_adapter(bridge_config) 

171 

172 driver = VLLMDriver( 

173 llm=llm, 

174 adapter=adapter, 

175 tokenizer=tokenizer, 

176 overlay=overlay, 

177 hf_config=hf_config, 

178 max_num_batched_tokens=max_num_batched_tokens, 

179 enable_batching=enable_batching, 

180 ) 

181 bridge = RemoteBridge(adapter=adapter, tokenizer=tokenizer, driver=driver) 

182 _log_hook_summary(model_name, architecture, driver) 

183 return bridge

186def _reject_locked_overrides(vllm_kwargs: Dict[str, Any]) -> None: 

187 for key, locked in _LOCKED_KWARGS.items(): 

188 if key in vllm_kwargs and vllm_kwargs[key] != locked: 

189 raise ValueError( 

190 f"boot_vllm forces {key}={locked}; caller passed {key}={vllm_kwargs[key]}. " 

191 "Multi-device / continuous batching / vLLM-owned tokenizer are unsupported." 

192 )

195def _dtype_from_hf_config(hf_config: Any) -> torch.dtype: 

196 raw = getattr(hf_config, "torch_dtype", None) 

197 if isinstance(raw, torch.dtype): 

198 return raw 

199 if isinstance(raw, str): 

200 return getattr(torch, raw, torch.float16) 

201 return torch.float16
